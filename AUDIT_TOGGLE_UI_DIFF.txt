+/* ── audit box builder ("Why this answer?") ──────────────────────────────
+   Renders one or more audit entries (intent/parameters/qualifying_pool_size/
+   routing_method/data_as_of) as a clean key-value list, not raw JSON. A
+   section can carry multiple audit entries (one per underlying compute
+   call, e.g. the Hustle section bundles 4 different functions) — each
+   renders as its own labeled block, separated by a hairline, rather than
+   collapsing distinct thresholds into one misleading summary. Entries that
+   are null (e.g. a section with no data for the given season) are skipped
+   silently — there is nothing to audit for those. ── */
+function formatAuditDate(iso) {
+  if (!iso) return null;
+  const d = new Date(iso);
+  if (isNaN(d)) return iso;
+  return d.toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' });
+}
+
+function formatAuditParams(params) {
+  const entries = Object.entries(params || {});
+  if (entries.length === 0) return '(none)';
+  return entries.map(([k, v]) => `${k}=${v}`).join(', ');
+}
+
+function buildAuditKV(key, value) {
+  const row = document.createElement('div');
+  row.className = 'audit-kv';
+  const k = document.createElement('span');
+  k.className = 'audit-key';
+  k.textContent = key;
+  const v = document.createElement('span');
+  v.className = 'audit-val';
+  v.textContent = value;
+  row.appendChild(k);
+  row.appendChild(v);
+  return row;
+}
+
+function buildAuditBox(auditList) {
+  const entries = (Array.isArray(auditList) ? auditList : [auditList]).filter(Boolean);
+  if (entries.length === 0) return null;
+
+  const box = document.createElement('div');
+  box.className = 'audit-box';
+
+  entries.forEach(a => {
+    const entry = document.createElement('div');
+    entry.className = 'audit-entry';
+    entry.appendChild(buildAuditKV('Intent', a.intent || '—'));
+    entry.appendChild(buildAuditKV('Parameters', formatAuditParams(a.parameters)));
+    entry.appendChild(buildAuditKV(
+      'Qualifying pool',
+      a.qualifying_pool_size != null ? `${a.qualifying_pool_size} players` : '—'
+    ));
+    entry.appendChild(buildAuditKV('Routing', a.routing_method || '—'));
+    entry.appendChild(buildAuditKV('Data as of', formatAuditDate(a.data_as_of) || '—'));
+    box.appendChild(entry);
+  });
+
+  return box;
+}
+
+/* Returns {toggle, box} — caller appends both, in order, wherever the
+   toggle should sit. Mirrors the table-toggle open/close pattern exactly
+   (arrow rotation, label swap, .visible class) so the two collapsibles
+   behave identically. */
+function buildAuditToggle(auditList) {
+  const box = buildAuditBox(auditList);
+  if (!box) return null;
+
+  const toggle = document.createElement('button');
+  toggle.className = 'audit-toggle';
+  toggle.innerHTML = '<span class="arrow">▶</span> <span class="toggle-label">Why this answer?</span>';
+
+  toggle.addEventListener('click', () => {
+    const open = box.classList.toggle('visible');
+    toggle.classList.toggle('open', open);
+  });
+
+  return { toggle, box };
+}
